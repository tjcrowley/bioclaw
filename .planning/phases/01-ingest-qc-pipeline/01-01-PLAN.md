```python
def tiny_mtx_dir(tmp_path) -> Path:
    """Pytest fixture. Writes a tiny 10x MEX-format directory
    (matrix.mtx.gz, barcodes.tsv.gz, features.tsv.gz) and returns its Path.
    ~15-20 genes x ~30-50 cells. Includes:
      - at least 2 genes prefixed 'MT-' (mitochondrial)
      - exactly one duplicated gene symbol (two different Ensembl IDs, same symbol)
      - at least one all-zero cell (barcode with zero counts across all genes)
      - at least one all-zero gene (gene with zero counts across all cells)
    """

def tiny_h5_file(tmp_path) -> Path:
    """Pytest fixture. Writes a tiny Cell Ranger-format HDF5
    feature-barcode matrix file (/matrix/{data,indices,indptr,shape,barcodes},
    /matrix/features/{id,name,feature_type,genome}) and returns its Path.
    Same gene/cell characteristics as tiny_mtx_dir (independently constructed,
    same shape is fine but not required to be identical).
    """

def synthetic_adata() -> AnnData:
    """Pytest fixture. Returns an in-memory AnnData with raw integer counts
    already in .X (sparse csr_matrix), NOT yet through loaders.py or the
    counts-layer contract. ~20 genes x ~50 cells. Includes:
      - at least 3 genes prefixed 'MT-' with meaningfully higher counts on a
        subset of cells (so pct_counts_mt varies enough to test threshold filtering)
      - at least one all-zero cell and one all-zero gene
      - enough spread in per-cell gene counts that a min_genes_per_cell
        threshold change visibly changes how many cells are filtered
    Used directly by QC and Store plans — they do NOT need loaders.py.
    """
```
