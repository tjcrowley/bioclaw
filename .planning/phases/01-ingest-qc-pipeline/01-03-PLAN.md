```python
# ingest/qc.py
@dataclass
class QCConfig:
    min_genes_per_cell: int = 200
    min_cells_per_gene: int = 3
    max_pct_mt: float | None = 20.0
    doublet_action: Literal["flag", "filter"] = "flag"

def run(adata: AnnData, cfg: QCConfig | None = None) -> AnnData:
    """Computes pct_counts_mt/n_genes_by_counts/total_counts/doublet_score/
    predicted_doublet on adata.obs; filters by cfg thresholds; writes
    adata.uns['qc'] = {config, n_cells_before, n_cells_after, removed_by_reason}.
    Never crashes on an empty result -- see edge case behavior below."""
```
