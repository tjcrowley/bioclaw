```python
# ingest/store.py
class DatasetStore:
    def __init__(self, root: str | Path = "data"):
        """root/ holds one .h5ad file per (name, version) plus root/registry.sqlite."""

    def save(self, name: str, adata: AnnData, source_path: str | None = None,
              qc_config: dict | None = None) -> int:
        """Writes adata to root/{name}/v{N}.h5ad where N = (latest existing
        version for this name) + 1 (starts at 1). Records name, version,
        path, created_at (UTC ISO8601), source_path, qc_config (JSON) in
        the SQLite registry. Returns the new version number (int)."""

    def load(self, name: str, version: int | None = None, backed: str | None = None) -> AnnData:
        """version=None loads the latest version for `name`. Raises KeyError
        if `name` (or that specific version) doesn't exist. `backed` passes
        through to anndata.read_h5ad's `backed=` kwarg (None by default --
        not exercised by this phase's data sizes, but the parameter must
        exist now so later phases don't require a signature change)."""

    def list(self, name: str | None = None) -> list[dict]:
        """Returns registry records (as dicts) for `name`, or all records if
        name is None, ordered by name then version."""
```
