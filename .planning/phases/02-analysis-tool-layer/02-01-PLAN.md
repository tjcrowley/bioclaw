```python
# analysis/summary.py
from dataclasses import dataclass, field

@dataclass
class PreprocessSummary:
    dataset_id: str | None       # "{name}@{version}" -- set by pipeline.py after store.save(); None from low-level preprocess()
    n_cells: int
    n_genes_total: int
    n_hvg: int
    n_pcs_computed: int
    variance_ratio_top10: list[float]   # first <=10 PCA components only, never the full array
    config: dict

@dataclass
class ClusterSummary:
    dataset_id: str | None
    n_clusters: int
    cluster_sizes: dict[str, int]      # {"0": 142, "1": 89, ...} -- O(n_clusters), never O(n_cells)
    resolution: float
    umap_computed: bool
    config: dict

@dataclass
class DEGeneResult:
    gene: str
    score: float
    pval: float
    pval_adj: float
    logfoldchange: float
    pct_group: float | None
    pct_rest: float | None

@dataclass
class DESummary:
    dataset_id: str | None
    groupby: str
    group1: str
    group2: str | None          # None means "rest"
    method: str                 # "wilcoxon"
    n_genes_tested: int
    n_significant: int          # count where pval_adj < 0.05
    top_genes: list[DEGeneResult]   # capped, e.g. top 25 by pval_adj -- never the full ranked array
```

`dataset_id` is `None` when a dataclass is built by a low-level module function (`preprocess()`, `cluster()`, `differential_expression()`), and is filled in by `analysis/pipeline.py` (Wave 2) once `store.save()` returns the new version -- mirrors how `ingest/qc.py`'s `QCConfig` is resolved independently of the store, then logged by `ingest/pipeline.py`.
