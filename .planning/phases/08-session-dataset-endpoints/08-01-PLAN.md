<objective>
Close the `session_id` plumbing gap identified in 08-RESEARCH.md (Pattern 1/Pitfall 1) and give `SessionMemory` the ability to enumerate every session that has started (Pattern 2/Pitfall 2), then expose that as `GET /api/sessions` / `GET /api/sessions/{session_id}` and make `POST /api/ask` actually honor an incoming `session_id` instead of silently discarding it.

Purpose: API-03 ("list existing sessions and resume a session by ID") is currently structurally impossible — `AskRequest.session_id` exists but is never read by `main.py::ask()`, `ask_question()` has no `session_id` parameter at all, and `SessionMemory` has no way to enumerate sessions (only dataset-reference rows written incidentally by a tool call). This plan closes both gaps additively, without touching the existing `session_memory` table, its tests, or any other already-passing test suite.
Output: A working `GET /api/sessions` (list) and `GET /api/sessions/{session_id}` (detail/existence-check) API, and a `POST /api/ask` that genuinely resumes a session when given a known `session_id`.
</objective>

```python
_CREATE_SESSIONS_TABLE_SQL = """
CREATE TABLE IF NOT EXISTS sessions (
    session_id TEXT PRIMARY KEY,
    created_at TEXT NOT NULL,
    last_active_at TEXT NOT NULL
)
"""

class SessionMemory:
    def __init__(self, root: str | Path = "agent/memory.sqlite"):
        self.path = Path(root)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self._connect() as conn:
            conn.execute(_CREATE_TABLE_SQL)
            conn.execute(_CREATE_SESSIONS_TABLE_SQL)   # NEW

    # ... existing record()/recent_datasets() unchanged ...

    def touch(self, session_id: str) -> None:
        now = datetime.now(timezone.utc).isoformat()
        with self._connect() as conn:
            conn.execute(
                "INSERT INTO sessions (session_id, created_at, last_active_at) "
                "VALUES (?, ?, ?) "
                "ON CONFLICT(session_id) DO UPDATE SET last_active_at = excluded.last_active_at",
                (session_id, now, now),
            )
            conn.commit()

    def list_sessions(self, limit: int = 50) -> list[dict]:
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT session_id, created_at, last_active_at FROM sessions "
                "ORDER BY last_active_at DESC, rowid DESC LIMIT ?",
                (limit,),
            ).fetchall()
        return [
            {
                "session_id": r[0],
                "created_at": r[1],
                "last_active_at": r[2],
                "recent_datasets": self.recent_datasets(r[0]),
            }
            for r in rows
        ]

    def session_exists(self, session_id: str) -> bool:
        with self._connect() as conn:
            row = conn.execute(
                "SELECT 1 FROM sessions WHERE session_id = ?", (session_id,)
            ).fetchone()
        return row is not None
```
