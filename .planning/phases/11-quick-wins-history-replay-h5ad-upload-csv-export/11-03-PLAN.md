<objective>
Add a CSV export capability so researchers can download cluster assignments, DE results, and annotation data from any analyzed dataset. This plan adds the GET /api/export/csv backend endpoint (returns a ZIP of three CSVs), adds a frontend download button wired to trigger the browser's native file-save flow, and covers the endpoint with unit tests.

Purpose: Researchers currently have no way to get their results out of BioClaw without digging into the SQLite/h5ad files directly. CSV export is the minimum viable data egress.
Output: GET /api/export/csv endpoint, exportCsv() JS function, download button in index.html/main.js, tests/test_webapp_export.py.

export function exportCsv(datasetId) {
    const url = `/api/export/csv?dataset_id=${encodeURIComponent(datasetId)}`;
    const a = document.createElement('a');
    a.href = url;
    a.download = '';  // browser uses Content-Disposition filename
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
}

    Then implement GET /api/export/csv in webapp/backend/main.py:

    ```python
    import io
    import zipfile
    import pandas as pd
    import scanpy as sc
    from fastapi.responses import StreamingResponse
    from ingest.store import DatasetStore

    @app.get("/api/export/csv", dependencies=[Depends(require_password)])
    async def export_csv(dataset_id: str) -> StreamingResponse:
        """EXPORT-01: Download cluster assignments, DE table, and annotation results
        for the named dataset as a ZIP of three CSV files."""
        # Parse dataset_id format: "name@version" or "name@(latest)"
        parts = dataset_id.split("@", 1)
        if len(parts) != 2:
            raise HTTPException(status_code=422, detail="dataset_id must be name@version (e.g. mydata@1)")
        name, version_str = parts
        version = int(version_str) if version_str.isdigit() else None

        store = DatasetStore(root=agent_tools.STORE_ROOT)
        try:
            adata = store.load(name, version)
        except KeyError as exc:
            raise HTTPException(status_code=404, detail=str(exc)) from exc

        buf = io.BytesIO()
        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
            # 1. Cluster assignments (leiden)
            if "leiden" in adata.obs.columns:
                clusters_df = adata.obs[["leiden"]].reset_index()
                clusters_df.columns = ["cell_barcode", "cluster"]
                zf.writestr("clusters.csv", clusters_df.to_csv(index=False))
            else:
                zf.writestr("clusters.csv", "cell_barcode,cluster\n(no cluster data)\n")

            # 2. DE table (rank_genes_groups)
            if "rank_genes_groups" in adata.uns:
                groups = list(adata.uns["rank_genes_groups"]["names"].dtype.names)
                de_frames = []
                for g in groups:
                    df = sc.get.rank_genes_groups_df(adata, group=g)
                    df.insert(0, "cluster", g)
                    de_frames.append(df)
                de_csv = pd.concat(de_frames, ignore_index=True) if de_frames else pd.DataFrame()
                zf.writestr("de_genes.csv", de_csv.to_csv(index=False))
            else:
                zf.writestr("de_genes.csv", "cluster,names,scores,pvals,pvals_adj,logfoldchanges\n(no DE data)\n")

            # 3. Annotations (from adata.uns — populated when annotate() has been called)
            ann_lines = ["cluster,method,label,confidence,ontology_term_id"]
            # NOTE: Read annotation/pipeline.py to confirm the exact uns key before finalising this block.
            # Research guidance says adata.uns["annotation"]; verify and correct if different.
            ann_data = adata.uns.get("annotation", {})
            for call in ann_data.get("fm_calls", []):
                ann_lines.append(
                    f"{call.get('cluster','')},fm,{call.get('label','')},{call.get('confidence','')},{call.get('ontology_term_id','')}"
                )
            for call in ann_data.get("baseline_calls", []):
                ann_lines.append(
                    f"{call.get('cluster','')},baseline,{call.get('label','')},{call.get('confidence','')},{call.get('ontology_term_id','')}"
                )
            if len(ann_lines) == 1:
                ann_lines.append("(no annotation data)")
            zf.writestr("annotations.csv", "\n".join(ann_lines) + "\n")

        buf.seek(0)
        safe_name = name.replace("/", "_").replace(" ", "_")
        return StreamingResponse(
            buf,
            media_type="application/zip",
            headers={"Content-Disposition": f'attachment; filename="{safe_name}_export.zip"'},
        )
    ```
