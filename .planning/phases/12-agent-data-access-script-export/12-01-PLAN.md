```python
def ingest_10x(path, name, qc_config: qc.QCConfig | None = None,
               store_root="data") -> str:
    adata = loaders.load(path)
    contract.set_counts_layer(adata)          # INGEST-02, before .X is touched
    qc_config = qc_config or qc.QCConfig()
    adata = qc.run(adata, qc_config)          # QC-01 + QC-02
    contract.set_counts_layer(adata)          # re-freeze against filtered shape
    store = DatasetStore(root=store_root)
    version = store.save(name, adata, source_path=str(path),
                         qc_config=asdict(qc_config))
    return f"{name}@{version}"
```

```python
                         obs_value_filter: str) -> str:

    'cellxgene-census:{census_version}:{organism}:{obs_value_filter}'"""

def ingest_from_anndata(adata, name, source_path: str,
                        qc_config: QCConfig | None = None,
                        store_root="data") -> str:
    """In-memory equivalent of ingest_10x for a PRE-BUILT AnnData (census
    fetch result). Runs set_counts_layer -> qc.run -> set_counts_layer ->
    store.save(source_path=source_path). Returns '{name}@{version}'.
    No temp file, no loaders.load()."""
```
